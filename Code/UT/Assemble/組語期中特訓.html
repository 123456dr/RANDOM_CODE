  cmp eax, -1

find_same:

find_greater:
  INVOKE printf, ADDR showBig

作業4-2 四則測驗（核心：出題、四路分支、記錄錯題、等第）

mnR dword 1
num dword 1
cntQ dword 1
op dword 1
numA dword ?
numB dword ?
ans dword ?
ansInput dword ?
correct dword 0
wrong dword 0
wCnt dword 4 DUP(0)
wA dword 50 DUP(0)
score dword 0
fmtAdd BYTE "%d) %d + %d = ", 0
fmtSub BYTE "%d) %d - %d = ", 0
fmtMul BYTE "%d) %d * %d = ", 0
fmtDiv BYTE "%d) %d / %d = ", 0
qFmt dword OFFSET fmtAdd, OFFSET fmtSub, OFFSET fmtMul, OFFSET fmtDiv
gradeA BYTE "A",0
gradeB BYTE "B",0
gradePass BYTE "PASS",0
gradeFail BYTE "FAIL",0
msgRes BYTE "score %d %s",10,0
; ---- .code ----
  mov mxR, 9            ; (mxR、mnR、num 由 scanf 讀入)
q_loop:
  mov eax, cntQ
  cmp eax, num
  jg q_end
  call rand              ; op = rand % 4
  xor edx, edx
  mov ecx, 4
  div ecx
  mov op, edx
  call rand              ; numA = mnR ~ mxR
  mov ecx, mxR
  sub ecx, mnR
  inc ecx
  xor edx, edx
  div ecx
  add edx, mnR
  mov numA, edx
  mov numB, 3            ; (numB 同上產生)
  cmp op, 0
  je do_add
  cmp op, 1
  je do_sub
  cmp op, 2
  je do_mul
do_div:                  ; 先定商，A = B×商
  mov eax, mxR
  xor edx, edx
  div numB
  mov ebx, eax           ; 商最大 = mxR/B
  call rand
  xor edx, edx
  div ebx
  inc edx
  mov ans, edx
  mov eax, numB
  mul edx
  mov numA, eax
  jmp ask
do_add:
  mov eax, numA
  add eax, numB
  mov ans, eax
  jmp ask                ; 都跳到匯合點
do_sub:
  mov eax, numA
  cmp eax, numB
  jge sub_ok             ; 不可小減大
  mov ebx, numB
  mov numA, ebx
  mov numB, eax
sub_ok:
  mov eax, numA
  sub eax, numB
  mov ans, eax
  jmp ask
do_mul:
  mov eax, numA
  mul numB
  mov ans, eax
ask:
  mov ebx, op            ; 用 op 挑格式字串
  INVOKE printf, qFmt[ebx*4], cntQ, numA, numB
  INVOKE scanf, ADDR formatIn, ADDR ansInput
  mov eax, ansInput
  cmp eax, ans
  jne q_wrong
  inc correct
  jmp q_next
q_wrong:
  mov ebx, wrong         ; 存到第 wrong 格
  mov eax, numA
  mov wA[ebx*4], eax     ; (wB wOp wAns 同法)
  inc wrong
  mov ebx, op
  inc wCnt[ebx*4]        ; 該類錯 +1
q_next:
  inc cntQ
  jmp q_loop
q_end:
  mov eax, correct       ; 分數 = 對×100÷題數
  mov ecx, 100
  mul ecx
  xor edx, edx
  div num
  mov score, eax
  mov ebx, OFFSET gradeFail ; 由差往好升級
  cmp score, 60
  jl show
  mov ebx, OFFSET gradePass
  cmp score, 80
  jl show
  mov ebx, OFFSET gradeB
  cmp score, 90
  jl show
  mov ebx, OFFSET gradeA
show:
  INVOKE printf, ADDR msgRes, score, ebx

範例：字串比對（repe cmpsb）、用堆疊反轉

s1 BYTE "Hello",0
s2 BYTE "Hello",0
nm BYTE "Taipei",0
nmLen = ($ - nm) - 1
fmtC BYTE "%d %s",10,0
; ---- .code ----
  mov esi, OFFSET s1
  mov edi, OFFSET s2
  mov ecx, LENGTHOF s1
  cld                    ; 往後比
  repe cmpsb             ; 一樣就繼續
  mov ebx, 0
  jne cDiff
  mov ebx, 1             ; 1 = 相同
cDiff: mov ecx, nmLen
  mov esi, 0
R1: movzx eax, nm[esi]   ; push 不能推 8 位
  push eax
  inc esi
  loop R1
  mov ecx, nmLen
  mov esi, 0
R2: pop eax              ; 反序拿回
  mov nm[esi], al
  inc esi
  loop R2
  INVOKE printf, OFFSET fmtC, ebx, OFFSET nm

寫完檢查：call 會弄壞 eax ecx edx · loop 用 ecx · mem,mem 不行 · ADDR 只在 INVOKE · esi 是位址→[esi]，是索引→arr[esi*4] · 除法前清 edx · 每個分支都 jmp · push/pop 成對 · 最後 ExitProcess。
